1. Quick Start - Simple Interface

<!-- Create a container -->
<div id="bluetooth-interface"></div>

        HiFiBerryHelpers.createPairingInterface(
            'bluetooth-interface', 
            'http://192.168.1.100/api/btaudio'  // Your HiFiBerry IP via nginx
        );
</script>

3. Custom Integration Example

            
For custom web applications, use the API directly:

            
            
// Custom integration
const btAPI = new HiFiBerryBluetoothAPI('http://192.168.1.100/api/btaudio');

// Basic pairing workflow
async function startPairing() {
    try {
        // Generate a random PIN
        const pinResult = await btAPI.generateRandomPin();
        console.log('Generated PIN:', pinResult.pin);
        
        // Enable pairing mode for 5 minutes
        await btAPI.makeDiscoverable(300);
        
        // Show PIN to user
        alert(`Pairing enabled! Use PIN: ${pinResult.pin}`);
        
        // Monitor for new devices
        const devicePoller = btAPI.startDevicePolling((error, devices) => {
            if (error) {
                console.error('Device polling error:', error);
                return;
            }
            
            console.log('Paired devices:', devices.devices);
            updateDeviceList(devices.devices);
        });
        
        // Stop polling after 5 minutes
        setTimeout(() => {
            devicePoller.stop();
            btAPI.hideFromDiscovery();
        }, 300000);
        
    } catch (error) {
        console.error('Pairing error:', error);
        alert('Error starting pairing: ' + error.message);
    }
}

// Check system status
async function checkBluetoothStatus() {
    try {
        const status = await btAPI.getStatus();
        const audio = await btAPI.checkAudio();
        
        return {
            bluetooth: status,
            audio: audio,
            ready: status.adapter.powered && audio.backend !== 'none'
        };
    } catch (error) {
        console.error('Status check error:', error);
        return { ready: false, error: error.message };
    }
}

// React component example
function BluetoothControl({ hifiberryIP }) {
    const [api] = useState(() => new HiFiBerryBluetoothAPI(`http://${hifiberryIP}:8080`));
    const [status, setStatus] = useState(null);
    const [devices, setDevices] = useState([]);
    
    useEffect(() => {
        const statusPoller = api.startStatusPolling((error, status) => {
            if (!error) setStatus(status);
        });
        
        const devicePoller = api.startDevicePolling((error, devices) => {
            if (!error) setDevices(devices.devices);
        });
        
        return () => {
            statusPoller.stop();
            devicePoller.stop();
        };
    }, [api]);
    
    const handleStartPairing = async () => {
        try {
            await api.enablePairing({ timeout: 120 });
        } catch (error) {
            alert('Error: ' + error.message);
        }
    };
    
    return (
        <div>
            {status && (
                <div>
                    <h3>{status.adapter.name}</h3>
                    <p>Status: {status.adapter.discoverable ? 'Discoverable' : 'Hidden'}</p>
                    <button onClick={handleStartPairing}>Start Pairing</button>
                </div>
            )}
            <ul>
                {devices.map(device => (
                    <li key={device.address}>
                        {device.name} ({device.connected ? 'Connected' : 'Paired'})
                    </li>
                ))}
            </ul>
        </div>
    );
}

        


        

            
4. Configuration

            
API endpoints and configuration options:

            
            
Available Endpoints:

            
    
                
  • GET /api/status - Get Bluetooth adapter status
    • 
                
  • GET /api/pairing/pin - Get current PIN
    • 
                
  • POST /api/pairing/pin/generate - Generate random PIN
    • 
                
  • POST /api/pairing/pin/set - Set custom PIN
    • 
                
  • POST /api/discoverable/start - Make discoverable
    • 
                
  • POST /api/discoverable/stop - Hide from discovery
    • 
                
  • GET /api/devices/paired - List paired devices
    • 
                
  • DELETE /api/devices/{mac} - Remove specific device
    • 
                
  • DELETE /api/devices/all - Remove all devices
    • 
                
  • GET /api/audio/check - Check audio backend status
    • 
                
  • POST /api/audio/setup - Setup audio profiles
    • 
            


            
Installation:

            
# Install the web API package
sudo apt install hifiberry-btaudio-web

# Start the web service
sudo systemctl start hifiberry-btaudio-web

# Enable auto-start
sudo systemctl enable hifiberry-btaudio-web


            
Security Considerations:

            
    
                
  • The API runs on port 8080 by default
    • 
                
  • Use firewall rules to restrict access if needed
    • 
                
  • Consider using HTTPS proxy for production
    • 
                
  • PIN codes are stored securely in /etc/hifiberry/btaudio.conf
